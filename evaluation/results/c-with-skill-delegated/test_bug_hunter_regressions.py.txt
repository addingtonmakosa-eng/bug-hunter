import copy

import pytest

import app
from transaction_service import TransactionService
from utils import apply_discount, is_valid_transaction, round_money, validate_transaction


def tx(transaction_id, amount, currency="USD", description=""):
    return {"id": transaction_id, "amount": amount, "currency": currency, "description": description}


def loaded(*records):
    svc = TransactionService()
    svc.load(list(records))
    return svc


@pytest.fixture
def service():
    return loaded(tx("A", 100.0), tx("B", 50.0), tx("A", 100.0), tx("C", -30.0))


def cli(capsys, *args):
    app.main(list(args))
    return capsys.readouterr().out.splitlines()


# --- BH-001: find_duplicates ------------------------------------------------


def test_find_duplicates_returns_every_later_repeat_in_order():
    svc = loaded(tx("A", 10.0), tx("B", 20.0), tx("A", 30.0), tx("C", 40.0), tx("A", 50.0))
    assert svc.find_duplicates() == [tx("A", 30.0), tx("A", 50.0)]


def test_find_duplicates_ignores_invalid_records():
    svc = loaded(tx("A", 0), tx("A", 10.0), tx("A", 20.0), tx("", 5.0), tx("", 6.0))
    assert svc.find_duplicates() == [tx("A", 20.0)]


def test_find_duplicates_does_not_raise_on_malformed_invalid_records():
    svc = loaded({"amount": 5.0}, tx(["x"], 7.0), tx(["x"], 8.0), tx("A", 1.0), tx("A", 2.0))
    assert svc.find_duplicates() == [tx("A", 2.0)]


def test_find_duplicates_keeps_invalid_records_in_the_store():
    svc = loaded(tx("A", 100.0), tx("", 25.0), tx("A", 100.0))
    before = svc.all_transactions()
    svc.find_duplicates()
    assert svc.count() == 3
    assert svc.all_transactions() == before


def test_find_duplicates_does_not_modify_record_contents():
    svc = loaded(tx("A", 100.0), tx("", 25.0), tx("A", 100.0))
    before = copy.deepcopy(svc.all_transactions())
    svc.find_duplicates()
    assert svc.all_transactions() == before
    assert svc.count() == 3


def test_find_duplicates_with_no_repeats_is_empty():
    assert loaded(tx("A", 1.0), tx("B", 2.0)).find_duplicates() == []
    assert TransactionService().find_duplicates() == []


def test_find_duplicates_gives_the_same_answer_twice(service):
    first = service.find_duplicates()
    second = service.find_duplicates()
    assert first == second == [tx("A", 100.0)]
    assert service.count() == 4


def test_cli_summary_duplicate_count(capsys):

def test_cli_duplicates_prints_one_line(capsys):

    assert "TXN-1002" in lines[0]

    assert apply_discount(200, 100) == 0.0

# --- BH-006: currency format ------------------------------------------------


def test_validate_rejects_lowercase_currency():
    assert len(validate_transaction(tx("A", 10.0, currency="usd"))) == 1
    assert is_valid_transaction(tx("A", 10.0, currency="usd")) is False


@pytest.mark.parametrize(
    "currency", ["Usd", "US1", "123", "U$D", "U D", "   ", "ÄÖÜ", "ＵＳＤ"]
)
def test_validate_rejects_three_character_currency_that_is_not_uppercase_ascii(currency):
    assert len(validate_transaction(tx("A", 10.0, currency=currency))) == 1


@pytest.mark.parametrize("currency", ["USD", "EUR", "JPY", "ZZZ"])
def test_validate_accepts_any_three_uppercase_ascii_letters(currency):
    assert validate_transaction(tx("A", 10.0, currency=currency)) == []


@pytest.mark.parametrize("currency", ["us", "", "USDX", "USD\n"])
def test_validate_wrong_length_currency_adds_exactly_one_problem(currency):
    assert len(validate_transaction(tx("A", 10.0, currency=currency))) == 1


@pytest.mark.parametrize("currency", [None, 840, ["U", "S", "D"], b"USD"])
def test_validate_non_string_currency_adds_one_problem_without_raising(currency):
    assert len(validate_transaction(tx("A", 10.0, currency=currency))) == 1


def test_service_treats_lowercase_currency_record_as_invalid():
    svc = TransactionService()
    with pytest.raises(ValueError):
        svc.add_transaction(tx("A", 10.0, currency="usd"))
    assert svc.count() == 0

    svc2 = loaded(tx("A", 100.0), tx("B", 40.0, currency="usd"))
    assert svc2.total_valid_transactions() == 100.0
    assert svc2.valid_transactions() == [tx("A", 100.0)]

    assert isinstance(result, float)


def test_validate_accepts_large_int_that_fits_a_float():
    assert validate_transaction(tx("A", 10**308)) == []

)
def test_validate_rejects_bad_amounts_with_exactly_one_problem(amount):
    assert len(validate_transaction(tx("A", amount))) == 1

@pytest.mark.parametrize("value", [None, [], "A", 42, [("id", "A")]])
def test_validate_non_dictionary_returns_one_problem(value):
    assert len(validate_transaction(value)) == 1
    assert is_valid_transaction(value) is False


def test_validate_stops_after_missing_fields():
    assert validate_transaction({"id": "", "amount": 0}) == ["missing field: currency"]


def test_validate_reports_one_problem_per_broken_rule():
    assert len(validate_transaction(tx("", 0, currency="US"))) == 3
    assert len(validate_transaction(tx("   ", 10.0))) == 1
    assert len(validate_transaction(tx(123, 10.0))) == 1


def test_validate_accepts_int_refund_without_description():
    assert validate_transaction({"id": "A", "amount": -5, "currency": "USD"}) == []


def test_add_transaction_stores_a_copy():
    svc = TransactionService()
    record = tx("A", 10.0)
    svc.add_transaction(record)
    record["amount"] = 999.0
    assert svc.all_transactions() == [tx("A", 10.0)]


def test_all_and_valid_transactions_return_new_lists(service):
    service.all_transactions().clear()
    service.valid_transactions().clear()
    assert service.count() == 4
    assert service.all_transactions() == [tx("A", 100.0), tx("B", 50.0), tx("A", 100.0), tx("C", -30.0)]


def test_load_appends_without_validation():
    svc = loaded(tx("A", 100.0), tx("", 25.0))
    svc.load([tx("B", 40.0)])
    assert svc.count() == 3
    assert svc.all_transactions() == [tx("A", 100.0), tx("", 25.0), tx("B", 40.0)]
    assert svc.valid_transactions() == [tx("A", 100.0), tx("B", 40.0)]


def test_round_money_returns_a_two_place_float():
    assert round_money(10) == 10.0
    assert isinstance(round_money(10), float)
    assert round_money(1.234) == 1.23
    assert round_money(1.236) == 1.24


def test_cli_summary_counts_and_average(capsys):
    lines = cli(capsys, "summary")
    assert [lines[i].split()[-1] for i in (0, 1, 3)] == ["6", "5", "100.25"]
